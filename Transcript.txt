# Sync cib-pipeline jars (+poms) from CIB Artifactory -> Dogen Artifactory (p-6771)
#
# CI/CD variables (Settings > CI/CD > Variables, masked):
#   ARTIFACTORY_USER / ARTIFACTORY_PASSWORD           -> Dogen (target, deploy rights)
#   ARTIFACTORY_CIB_USER / ARTIFACTORY_CIB_PASSWORD   -> CIB (source, read rights)
#
# Schedule: CI/CD > Schedules > cron "0 */6 * * *" (every 6 hours)
# Manual:   CI/CD > Pipelines > Run pipeline  (optionally set LOOKBACK, e.g. 7d)
# Extra logging: set DEBUG=true when running manually (never prints credentials)

stages:
  - sync

variables:
  CIB_URL: "https://artifactory.cib.echonet/artifactory"
  DOGEN_URL: "https://artifactory-dogen.group.echonet/artifactory"
  LOOKBACK: "2d"                      # only artifacts modified in this window
  ARTIFACT_PATH: "com/bnpparibas/cib/gmiteng/cib-pipeline/*"
  DEBUG: "false"
  # source repo -> target repo
  SRC_SNAPSHOT: "gmiteng-maven-local-dev"
  DST_SNAPSHOT: "p-6771-maven-SNAPSHOT"
  SRC_RELEASE: "gmiteng-maven-local-release"
  DST_RELEASE: "p-6771-maven-RELEASE"

sync-cib-to-dogen:
  stage: sync
  # image: <your-standard-image>   # needs: curl, jq, sha1sum
  rules:
    - if: $CI_PIPELINE_SOURCE == "schedule"
    - if: $CI_PIPELINE_SOURCE == "web"
  resource_group: cib-dogen-sync
  timeout: 1h
  script:
    - |
      set -eu
      log()  { echo "[$(date -u +%H:%M:%S)] $*"; }
      dbg()  { [ "$DEBUG" = "true" ] && log "DEBUG: $*" || true; }

      CIB_AUTH="$ARTIFACTORY_CIB_USER:$ARTIFACTORY_CIB_PASSWORD"
      DOGEN_AUTH="$ARTIFACTORY_USER:$ARTIFACTORY_PASSWORD"
      WORK="$(mktemp -d)"
      TOTAL_COPIED=0; TOTAL_SKIPPED=0; TOTAL_FAILED=0

      # ---------- environment / config (no secrets) ----------
      log "===== ENVIRONMENT ====="
      log "Pipeline source : $CI_PIPELINE_SOURCE  | job: $CI_JOB_ID  | runner: ${CI_RUNNER_DESCRIPTION:-n/a}"
      log "CIB URL         : $CIB_URL"
      log "Dogen URL       : $DOGEN_URL"
      log "Lookback        : $LOOKBACK"
      log "Artifact path   : $ARTIFACT_PATH"
      log "Mapping         : $SRC_SNAPSHOT -> $DST_SNAPSHOT"
      log "Mapping         : $SRC_RELEASE -> $DST_RELEASE"
      log "CIB user        : ${ARTIFACTORY_CIB_USER:-<EMPTY>}"
      log "Dogen user      : ${ARTIFACTORY_USER:-<EMPTY>}"
      for V in ARTIFACTORY_CIB_USER ARTIFACTORY_CIB_PASSWORD ARTIFACTORY_USER ARTIFACTORY_PASSWORD; do
        eval "VAL=\${$V:-}"
        [ -n "$VAL" ] && log "Variable $V is set (length ${#VAL})" || { log "ERROR: variable $V is EMPTY/not set (check scope/protected branch)"; exit 1; }
      done
      for T in curl jq sha1sum; do
        command -v "$T" > /dev/null 2>&1 && log "Tool $T: $(command -v $T)" || { log "ERROR: tool '$T' missing in image"; exit 1; }
      done
      log "curl version    : $(curl --version | head -1)"

      # ---------- pre-flight checks ----------
      http_code() { # $1=auth $2=url
        curl -sS -o /dev/null -w '%{http_code}' --max-time 30 -u "$1" "$2" 2>"$WORK/curl.err" || { cat "$WORK/curl.err"; echo 000; }
      }
      log "===== PRE-FLIGHT CHECKS ====="
      log "CIB   ping (no auth)       : HTTP $(curl -sS -o /dev/null -w '%{http_code}' --max-time 30 "$CIB_URL/api/system/ping" 2>&1 || true)"
      log "Dogen ping (no auth)       : HTTP $(curl -sS -o /dev/null -w '%{http_code}' --max-time 30 "$DOGEN_URL/api/system/ping" 2>&1 || true)"
      log "CIB   auth on $SRC_SNAPSHOT : HTTP $(http_code "$CIB_AUTH" "$CIB_URL/api/storage/$SRC_SNAPSHOT")   (200=ok 401=bad creds 403=no permission 404=repo name wrong)"
      log "CIB   auth on $SRC_RELEASE  : HTTP $(http_code "$CIB_AUTH" "$CIB_URL/api/storage/$SRC_RELEASE")"
      log "Dogen auth on $DST_SNAPSHOT : HTTP $(http_code "$DOGEN_AUTH" "$DOGEN_URL/api/storage/$DST_SNAPSHOT")"
      log "Dogen auth on $DST_RELEASE  : HTTP $(http_code "$DOGEN_AUTH" "$DOGEN_URL/api/storage/$DST_RELEASE")"

      # ---------- sync ----------
      sync_repo() {
        SRC="$1"; DST="$2"; COPIED=0; SKIPPED=0; FAILED=0
        log "===== SYNC $SRC -> $DST (last $LOOKBACK) ====="

        QUERY="items.find({\"repo\":\"$SRC\",\"path\":{\"\$match\":\"$ARTIFACT_PATH\"},\"\$or\":[{\"name\":{\"\$match\":\"*.jar\"}},{\"name\":{\"\$match\":\"*.pom\"}}],\"modified\":{\"\$last\":\"$LOOKBACK\"}}).include(\"repo\",\"path\",\"name\",\"actual_sha1\")"
        log "AQL: $QUERY"

        HTTP=$(curl -sS --max-time 120 -u "$CIB_AUTH" -H "Content-Type: text/plain" \
          -X POST "$CIB_URL/api/search/aql" --data-binary "$QUERY" \
          -o "$WORK/result.json" -w '%{http_code}') || HTTP=000
        if [ "$HTTP" != "200" ]; then
          log "ERROR: AQL request failed, HTTP $HTTP. Response body:"
          head -c 2000 "$WORK/result.json" 2>/dev/null; echo
          exit 1
        fi

        COUNT=$(jq '.results | length' "$WORK/result.json")
        log "AQL returned $COUNT file(s)"
        dbg "first results: $(jq -c '.results[:3]' "$WORK/result.json")"

        jq -r '.results[] | "\(.path)/\(.name) \(.actual_sha1)"' "$WORK/result.json" > "$WORK/list.txt"
        while read -r FILE SHA1; do
          DST_SHA1=$(curl -sS --max-time 60 -u "$DOGEN_AUTH" "$DOGEN_URL/api/storage/$DST/$FILE" 2>/dev/null \
                     | jq -r '.checksums.sha1 // empty' 2>/dev/null || true)
          if [ "$DST_SHA1" = "$SHA1" ]; then
            dbg "SKIP (already in Dogen, sha1 $SHA1): $FILE"
            SKIPPED=$((SKIPPED+1)); continue
          fi

          DL=$(curl -sS --max-time 600 -u "$CIB_AUTH" -o "$WORK/file.bin" -w '%{http_code}' "$CIB_URL/$SRC/$FILE") || DL=000
          if [ "$DL" != "200" ]; then
            log "FAIL download HTTP $DL: $FILE"; FAILED=$((FAILED+1)); continue
          fi
          LOCAL_SHA1=$(sha1sum "$WORK/file.bin" | cut -d' ' -f1)
          if [ "$LOCAL_SHA1" != "$SHA1" ]; then
            log "FAIL checksum mismatch (expected $SHA1, got $LOCAL_SHA1): $FILE"; FAILED=$((FAILED+1)); continue
          fi

          UL=$(curl -sS --max-time 600 -u "$DOGEN_AUTH" -X PUT -H "X-Checksum-Sha1: $SHA1" \
                 -T "$WORK/file.bin" -o "$WORK/upload.out" -w '%{http_code}' "$DOGEN_URL/$DST/$FILE") || UL=000
          case "$UL" in
            200|201) log "COPIED ($(wc -c < "$WORK/file.bin") bytes): $FILE"; COPIED=$((COPIED+1));;
            *) log "FAIL upload HTTP $UL: $FILE  (401=creds 403=no deploy/overwrite permission 409=conflict)"
               head -c 500 "$WORK/upload.out"; echo
               FAILED=$((FAILED+1));;
          esac
        done < "$WORK/list.txt"

        log "RESULT $SRC: found=$COUNT copied=$COPIED skipped=$SKIPPED failed=$FAILED"
        TOTAL_COPIED=$((TOTAL_COPIED+COPIED)); TOTAL_SKIPPED=$((TOTAL_SKIPPED+SKIPPED)); TOTAL_FAILED=$((TOTAL_FAILED+FAILED))
      }

      sync_repo "$SRC_SNAPSHOT" "$DST_SNAPSHOT"
      sync_repo "$SRC_RELEASE"  "$DST_RELEASE"

      log "===== SUMMARY: copied=$TOTAL_COPIED skipped=$TOTAL_SKIPPED failed=$TOTAL_FAILED ====="
      [ "$TOTAL_FAILED" -eq 0 ] || { log "Job failed: $TOTAL_FAILED file(s) not copied"; exit 1; }
